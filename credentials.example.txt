# ══════════════════════════════════════════════════════════════════════════════
#  CREDENTIALS TEMPLATE
# ══════════════════════════════════════════════════════════════════════════════
#
#  1. Copy this file:      cp credentials.example.txt credentials.txt
#  2. Fill in ONLY the section for the deployment you will use.
#  3. Run:                 node run.js
#
#  credentials.txt is listed in .gitignore, so your keys are never committed.
#  Keep this example file free of real values.
#
#  FORMAT
#    [section]        one section per deployment type
#    KEY = value      one setting per line; text after " #" is a comment
#    Don't quote values. A blank value means "not set".
#
#  The run.js menu option picks the section:
#    1 → [serverless]     2 → [cloud_hosted]     3 → [self_managed]
#
#  API KEY
#    Kibana → Stack Management → API Keys → Create API key.
#    Copy the "Encoded" value (one base64 string), not the id / api_key pair.
#    The key needs write access to logs-* and .alerts-security.alerts-*;
#    for a throwaway demo cluster, the default (owner's privileges) is simplest.
# ══════════════════════════════════════════════════════════════════════════════


# ──────────────────────────────────────────────────────────────────────────────
# [1] ELASTIC CLOUD SERVERLESS  (Security project)
# ──────────────────────────────────────────────────────────────────────────────
# Serverless has no Cloud ID. Copy both endpoints from the Elastic Cloud console:
#   your project → Manage → Application endpoints (Elasticsearch and Kibana).
# API key auth only.
# ──────────────────────────────────────────────────────────────────────────────
[serverless]
ELASTICSEARCH_URL     =                 # https://<project>.es.<region>.<csp>.elastic.cloud
KIBANA_URL            =                 # https://<project>.kb.<region>.<csp>.elastic.cloud
ELASTICSEARCH_API_KEY =
KIBANA_API_KEY        =                 # optional, reuses ELASTICSEARCH_API_KEY if blank
KIBANA_SPACE_ID       = default


# ──────────────────────────────────────────────────────────────────────────────
# [2] ELASTIC CLOUD HOSTED  (a deployment on cloud.elastic.co)
# ──────────────────────────────────────────────────────────────────────────────
# The Cloud ID is all you need for endpoints. run.js decodes it into the
# Elasticsearch and Kibana URLs. Find it on the deployment's overview page.
# Fill ELASTICSEARCH_URL / KIBANA_URL only to override the decoded values.
# ──────────────────────────────────────────────────────────────────────────────
[cloud_hosted]
ELASTICSEARCH_CLOUD_ID =                # <deployment-name>:<base64 string>
ELASTICSEARCH_API_KEY  =
ELASTICSEARCH_URL      =
KIBANA_URL             =
KIBANA_API_KEY         =
KIBANA_SPACE_ID        = default

# Basic-auth fallback, used only if ELASTICSEARCH_API_KEY is blank.
# Note: the demo-log ingest step accepts API keys only.
ELASTICSEARCH_USERNAME =
ELASTICSEARCH_PASSWORD =


# ──────────────────────────────────────────────────────────────────────────────
# [3] ELASTIC SELF-MANAGED  (on-premise / Docker / local)
# ──────────────────────────────────────────────────────────────────────────────
# The demo-log ingest step REQUIRES an API key, because it authenticates
# with "Authorization: ApiKey" only. Username/password works for the simulator.
#
# To create an API key on a self-managed cluster:
#   curl -u elastic:<password> -k -X POST "https://localhost:9200/_security/api_key" \
#     -H 'Content-Type: application/json' -d '{"name":"ai-soc-demo"}'
#   Use the "encoded" field from the response.
#
# TLS: point CA_CERT at your cluster's http_ca.crt (preferred), or set
#      VERIFY_SSL = false to skip certificate checks (lab use only).
# ──────────────────────────────────────────────────────────────────────────────
[self_managed]
ELASTICSEARCH_URL      = https://localhost:9200
KIBANA_URL             = http://localhost:5601
ELASTICSEARCH_API_KEY  =
ELASTICSEARCH_USERNAME = elastic
ELASTICSEARCH_PASSWORD =
KIBANA_API_KEY         =
KIBANA_SPACE_ID        = default
VERIFY_SSL             = true
CA_CERT                =
